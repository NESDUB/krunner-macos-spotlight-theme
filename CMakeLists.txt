cmake_minimum_required(VERSION 3.22)
project(krunner_spotlight VERSION 6.7.5 LANGUAGES CXX)
# This KRunner fork must be built against the matching Plasma 6.7.5 APIs.
# Do not advertise later Plasma/Qt versions as supported without rebuilding and testing.
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
set(QT_MAJOR_VERSION 6)
include(KDEInstallDirs6)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(ECMQmlModule)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Quick Widgets DBus)
find_package(KF6 REQUIRED COMPONENTS KIO I18n DBusAddons Crash Runner WindowSystem Config Service)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
find_package(PlasmaActivities REQUIRED)
find_package(LayerShellQt REQUIRED)
set(app_sources main.cpp view.cpp view.h x11windowscreenrelativepositioner.cpp x11windowscreenrelativepositioner.h)
qt_add_dbus_adaptor(app_sources dbus/org.kde.krunner.App.xml view.h View)
add_executable(krunner-spotlight ${app_sources})
ecm_add_qml_module(krunner-spotlight URI "org.kde.krunner.private.view" QML_FILES qml/RunCommand.qml qml/SpotlightResult.qml qml/GlassPanel.qml DEPENDENCIES org.kde.plasma.core)
target_link_libraries(krunner-spotlight PRIVATE Qt6::Quick Qt6::Widgets Qt6::DBus KF6::KIOGui KF6::I18n Plasma::PlasmaQuick Plasma::Activities KF6::DBusAddons KF6::Crash KF6::Runner KF6::WindowSystem KF6::ConfigCore KF6::Service LayerShellQt::Interface)
target_compile_definitions(krunner-spotlight PRIVATE PROJECT_VERSION="${PROJECT_VERSION}")
option(KRUNNER_SPOTLIGHT_DEVELOPER_PREVIEW "Build opt-in screenshot/diagnostic probes" OFF)
if(KRUNNER_SPOTLIGHT_DEVELOPER_PREVIEW)
    target_compile_definitions(krunner-spotlight PRIVATE KRUNNER_SPOTLIGHT_DEVELOPER_PREVIEW=1)
endif()

# No system-wide install rules: tools/manage.py deploys the built executable user-locally.
